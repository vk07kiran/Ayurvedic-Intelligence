CREATE TABLE Users (
    UserID        VARCHAR(50) PRIMARY KEY,
    Username      VARCHAR(100) UNIQUE NOT NULL,
    UserEmail     VARCHAR(150) UNIQUE NOT NULL,
    UserPassword  VARCHAR(255) NOT NULL,
    UserType      ENUM('Doctor', 'Patient') NOT NULL
);

CREATE TABLE Doctor (
    DoctorID                 VARCHAR(50) PRIMARY KEY,         -- FK to Users(UserID)
    DoctorProfilePic         VARCHAR(255),
    DoctorFirstName          VARCHAR(100) NOT NULL,
    DoctorLastName           VARCHAR(100),
    DoctorPhone              VARCHAR(20) UNIQUE,
    DoctorGender             VARCHAR(10),
    DoctorDOB                DATE,
    DoctorBio                TEXT,
    DoctorClinicName         VARCHAR(150),
    DoctorClinicAddress      TEXT,
    DoctorImages             TEXT,
    DoctorAddress1           VARCHAR(255),
    DoctorAddress2           VARCHAR(255),
    DoctorCity               VARCHAR(100),
    DoctorState              VARCHAR(100),
    DoctorCountry            VARCHAR(100),
    DoctorPostalCode         VARCHAR(20),
    DoctorFeePerHour         DECIMAL(10,2),
    DoctorSpecialization     VARCHAR(255),
    DoctorServices           TEXT,      -- comma-separated or JSON
    DoctorDegreeName         VARCHAR(150),
    DoctorCollege            VARCHAR(150),
    DoctorYearOfCompletion   YEAR,
    DoctorExperienceHospital VARCHAR(150),
    DoctorExperienceFromYear YEAR,
    DoctorExperienceToYear   YEAR,
    DoctorExperienceDesignation VARCHAR(150),
    DoctorAward              VARCHAR(150),
    DoctorAwardYear          YEAR,
    DoctorMemberships        TEXT,
    DoctorRegistrations      TEXT,
    DoctorRegistrationYear   YEAR,

    FOREIGN KEY (DoctorID) REFERENCES Users(UserID) ON DELETE CASCADE
);

CREATE TABLE Patient (
    PatientUserId                VARCHAR(50) PRIMARY KEY,                 -- FK from Users(UserID)
    PatientProfilePic            VARCHAR(255),
    PatientFirstName             VARCHAR(100) NOT NULL,
    PatientLastName              VARCHAR(100),
    PatientDOB                   DATE,
    PatientBloodGroup            VARCHAR(5),
    PatientMobile                VARCHAR(20) UNIQUE,
    PatientAddress               TEXT,
    PatientCity                  VARCHAR(100),
    PatientState                 VARCHAR(100),
    PatientZipCode               VARCHAR(20),
    PatientCountry               VARCHAR(100),

    PatientAppointedDoctorID     VARCHAR(50),                             -- FK from Doctor(DoctorID)

    PatientPrescriptionName      VARCHAR(255),
    PatientPrescriptionDate      DATE,
    PatientAppointmentDate       DATE,
    PatientPrescriptionDoctorName VARCHAR(150),
    PatientRecords               TEXT,

    PatientBillingDoctorName     VARCHAR(150),
    PatientBillingAmount         DECIMAL(10,2),
    PatientBillingDate           DATE,
    PatientBillingInvoiceNo      VARCHAR(50),

    FOREIGN KEY (PatientUserId) REFERENCES Users(UserID) ON DELETE CASCADE,
    FOREIGN KEY (PatientAppointedDoctorID) REFERENCES Doctor(DoctorID) ON DELETE SET NULL
);
