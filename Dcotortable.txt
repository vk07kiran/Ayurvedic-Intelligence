CREATE TABLE Doctor (
    DoctorProfilePic         VARCHAR(255),
    DoctorID                 int PRIMARY KEY,
    DoctorUsename            VARCHAR(100) UNIQUE NOT NULL,
    DoctorEmail              VARCHAR(150) UNIQUE NOT NULL,
    DoctorFirstName          VARCHAR(100) NOT NULL,
    DoctorLastName           VARCHAR(100),
    DoctorPassword           VARCHAR(255) NOT NULL,
    DoctorPhone              VARCHAR(20) UNIQUE,
    DoctorGender             VARCHAR(10),
    DoctorDOB                DATE,
    DoctorBio                TEXT,
    DoctorClinicName         VARCHAR(150),
    DoctorClinicAddress      TEXT,
    DoctorImages             TEXT,
    DoctorAddress1           VARCHAR(255),
    DoctorAddress2           VARCHAR(255),
    DoctorCity               VARCHAR(100),
    DoctorState              VARCHAR(100),
    DoctorCountry            VARCHAR(100),
    DoctorPostalCode         VARCHAR(20),
    DoctorFeePerHour         DECIMAL(10,2),
    DoctorSpecialization     VARCHAR(255),
    DoctorServices           TEXT, -- can store comma-separated services or use JSON
    DoctorDegreeName         VARCHAR(150),
    DoctorCollege            VARCHAR(150),
    DoctorYearOfCompletion   YEAR,
    DoctorExperienceHospital VARCHAR(150),
    DoctorExperienceFromYear YEAR,
    DoctorExperienceToYear   YEAR,
    DoctorExperienceDesignation VARCHAR(150),
    DoctorAward              VARCHAR(150),
    DoctorAwardYear          YEAR,
    DoctorMemberships        TEXT,
    DoctorRegistrations      TEXT,
    DoctorRegistrationYear   YEAR
);
